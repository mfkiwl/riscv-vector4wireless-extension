== Fast non-linear instruction implemenation and paramters
=== norm

. *Description*
+
normalization for the input data to [0.5~1).

. *Operation*
+
vs2 is 32bit fixpoint real number, vs1 is the fix point position, e.g. for a Q32.20 fixpoint value, it's fix point position number is 20.
+
----
x, E = norm(R_NL_TYPE, vs2, vs1)

if R_NL_TYPE == 4 || R_NL_TYPE == 6 || R_NL_TYPE == 7   % arctan/sin/cos: norm should be skipped
    x = vs2;
    E = 0;
else
    eNorm =  lsb(vs2); // take 32 bit-width data for example
    x = vs2 << eNorm;
    E = 31 - eNorm - vs1；

----

=== preProc

. *Description*
+
preprocess for non-linear instruction.

. *Operation*
+
----
x, s, E = preproc(R_NL_TYPE, x, E)

if R_NL_TYPE == 0       % reciprocal
    s = sign(x);        % if x>0 s=1; elsif x<0 s=-1; elsif x==0 s=0;
    x = abs(x);

elsif R_NL_TYPE == 1    % square root
    isOddE = alg_round_shift_right(E,1) - alg_shift_right(E,1);
    E   = E + isOddE;
    x   = x >> isOddE;

elsif R_NL_TYPE == 2    % reciprocal of square root
    isOddE = alg_round_shift_right(E,1) - alg_shift_right(E,1);
    E   = E + isOddE;
    x   = x >> isOddE;

elsif R_NL_TYPE == 3 || R_NL_TYPE == 5   % log2/log10 should be skipped

elsif R_NL_TYPE == 4    % arctan
    s = sign(x);        % if x>0 s=1; elsif x<0 s=-1; elsif x==0 s=0;
    x = abs(x);

elsif R_NL_TYPE == 6 || R_NL_TYPE == 7    % sin/cos should be skipped

----

=== findSeg

. *Description*
+
find which segment the input data belongs to in N-th segment group.

. *Operation*
+
----
segIdx = findSeg(x, R_SEG[N])  // N is segment group ID

for iSeg = R_SEG[N].nSegments        // segment number in one group
    lutIndex(iSeg) = x >=R_SEG[N].xStart(iSeg);
end
segIdx = sum(lutIndex);          // find which segment belong to
----


=== expAdjust

. *Description*
+
adjust the exponent for the output.

. *Operation*
+
----
e, E = ExpAdjust(R_NL_TYPE, E)

if R_NL_TYPE == 0       % reciprocal
    E =  -E + 1;
    if (E >= 0) e = 31 - E;
    else e = 31;

elsif R_NL_TYPE == 1    % square root
    E = (E>>1);
    if (E >= 0) e = 31 - E;
    else e = 31;   
   
elsif R_NL_TYPE == 2    % reciprocal of square root
    E = -(E>>1)+1;
    if (E >=  0) e = 31 - E;
    else e = 31;

elsif R_NL_TYPE == 3    % log2
    e = 26;

elsif R_NL_TYPE == 4    % arctan
    e = 28;

elsif R_NL_TYPE == 5    % log10
    e = 26;

elseif R_NL_TYPE == 6 || R_NL_TYPE == 7 % sin/cos
    e = 31;

----


=== postProc

. *Description*
+
postprocess for non-linear instruction.

. *Operation*
+
----
y = postproc(R_NL_TYPE, tmp, s, E)

if R_NL_TYPE == 0       % reciprocal
    y = s * tmp;
    if (E < 0) y = y >> -E;
    
elsif R_NL_TYPE == 1    % square root
    if (E < 0) y = y >> -E;

elsif R_NL_TYPE == 2    % reciprocal of square root 
    y = tmp;
    if (E < 0) y = y >> -E;

elsif R_NL_TYPE == 3    % log2
    if tmp>0 tmp=0;
    if tmp<1-2^31 tmp = 1-2^31  % tmp belong to range (-1,0)
    y = tmp>>5 + E<<26;         % final output Q(32.26)

elsif R_NL_TYPE == 4    % arctan
    y = s*tmp + 843314857;          % Q(32.30) const for pi/4
    y = alg_round_shift_right(y,2)  % Q(32.28)

elsif R_NL_TYPE == 5    % log10
    if tmp>0 tmp=0;
    if tmp<1-2^31 tmp = 1-2^31          % tmp belong to range (-1,0)
    y0 = tmp>>5 + E<<26;                % log2 output Q(32.26)
    y1 = y0*1292913986;                 % Q(32.32) for const log10(2)
    y =  alg_round_shift_right(y1, 32); % log10 final output

elsif R_NL_TYPE == 6 || R_NL_TYPE == 7  % sin/cos
    y = tmp;

----
